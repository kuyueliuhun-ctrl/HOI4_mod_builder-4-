# Archetype equipment follows the following format:
	# 原型装备遵循以下格式：
<pre>
	equipments = {
		<equipment> = {
			year = <int>					# Limits the equipment from appearing before the specified year. Optional
												# 限制设备生产年份。可选
			picture = <name>				# GFX reference used to define equipment picture in land-lease
												# 所参照的图片
			is_archetype = yes				# Specifies an entry as an archetype entry. All non-archetype entries inherit
												# 指定为原型条目。继承所有非原型条目
			is_buildable = no				# Prevents this equipment from being built.
												# 阻止制造装备
			active = yes					# Determines if this equipment is available without unlocking from a technology.
												# 确定使用此装备，而不需要解锁技术。
			type = <type>					# Internal type: what kind of unit can use this equipment
												# 类型：什么类型的单元可以使用此设备？
			group_by = <group>				# How the equipment is grouped in the production screen
												# 显示装备归类在生产界面上
			interface_category = <type>		# Which category the equipment appears in the production screen
												# 在生产界面上显示装备类别
		# Resources used to build this equipment
			# 用于建造此设备的资源
			resources = {
				<resource> = <amount>
			}
		
		# Modifiers the equipment uses
			#装备修改语句
			<modifiers> 
		}
	}
</pre>

# Regular equipment follows the following format:
	# 常规装备遵循以下格式
<pre>
	equipments = {
		<equipment> = {
			year = <int>					# Limits the equipment from appearing before the specified year. Optional
												# 限制设备生产年份。可选
			active = yes					# Determines if this equipment is available without unlocking from a technology.
												# 确定使用此装备，而不需要解锁技术。
			archetype = <equipment>			# Which archetype equipment this equipment inherits from.
												# 继承哪个原始装备。
			parent = <equipment>			# Which equipment is parent to this equipment (i.e. which does it supercede)
												# 哪个装备是源自这件装备。（也就是代替）
			priority = <int>				# Priority for usage over other equipment.
												# 优先于其他设备使用。（使用优先级）
			visual_level = <int>			# Image priority in production screen
												#生产界面中的优先级
		# Resources used to build this equipment
			# 用于建造此设备的资源
			resources = {
				<resource> = <amount>
			}
			
		# Modifiers the equipment uses
			# 装备修改语句
			<modifiers> 
		}
	}
</pre>

<pre>
	sub_units = {
		<unit name> = {
			sprite = <entity>			# Refers to <name>_entity in a gfx/entities/.*.asset file - Controls the unit used on the map
											# 指的是<name>_entity可控制地图上单位信息文件gfx/entities/.*.asset
		# Controls the map icon category used for the unit
			# 控制该单元使用的映射图标类别
			map_icon_category = <type>
			
			priority = <int>			# Controls which icon is used if multiple subunits of the same type exist
											# 如果存在相同类型的多个子单元，则使用哪个图标
			ai_priority = <int>
			
			active = yes / no			# Whether the unit is usable without explicit unlocking in technology
											# 是否通过技术解锁该单元
			cavalry = yes				# Used to define subunit as cavalry
											# 用于定义子单位为骑兵部队
			special_forces = yes		# Used to define subunit as special forces
											# 用于定义子单位为特种部队
			marines = yes				# Used to define subunit as marine
											# 用于定义子单位为海军陆战队
			mountaineers = yes			# Used to define subunit as mountaineers
											# 用于定义子单位为山地部队
			transport = <equipment>		# Sets the speed of subunit via equipment. Used for motorized/mechanized
											# 用于定义子单位为摩托化/机械化部队。速度通过装备设定
			
		# How the unit is grouped division-wise, i.e. appearing in the Support tab. Not used for air or naval units
			# 单元如何划分分组，即出现在支持选项卡中。不用于空军或海军单位
			group = <group> 
			
		# Internal types
			#类型
			type = {
				<type>
			}
			
		# Categories the unit belongs to.
			# 属于本单位的类别。
			categories = {
				<categories>
			}
			
		# Equipment that is required for this unit. Used when active = no
			# 本单位所需的装备。在运动时使用=否
			essential = {
				<equipment>
			}
			
		# Equipment needed to produce this unit. Unit won't be available until the nation has this equipment
			# 生产该单位所需的装备。除非国家有这种装备，否则单位将不能使用。
			need = {
				<equipment>
			}
			
		# Modifiers control the stats of the unit. More detail under the Stats section.
			# 单位统计信息修改语句。部分统计后更多细节
			<modifiers>
			
		# Modifiers that occur only in specific terrain
			# 仅发生在特定地形中的修改语句
			<terrain> = {
				<modifiers>
			}
		}
	}
</pre>

	supply_consumption = <float>

#####################################################################################################################################################

# The following list is all the valid modifiers for use in equipment and units:
	# 以下列表是装备和单位中使用的所有有效修改语句：

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Base（基础语句） >>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

<pre>
	max_organisation = 20		# Organisation - Organisation indicates combat readiness and how organized a unit is. A unit with no organisation can't fight or move effectively.
									# 组织度-组织度表示战备和组织方式。没有组织度的单位不能有效地战斗或行动。
	reliability = 0.9			# Reliability - The lower the reliability, the more likely the equipment will suffer random failure
									# 可靠性-可靠性越低，设备就越容易发生随机故障
	weight = 0.1				# Weight - How much the unit will weigh. Heavier units will require more transports to ship and perform invasions effectively.
									# 重量-单位重量多少。较重的单位将需要更多的运输船和有效地执行入侵。
	maximum_speed = 4			# Max Speed - How quickly this unit can traverse terrain under optimal circumtances, in kilometres per hour
									# 最高速度多快这个单位可以穿越地形下的最优条件，以公里每小时
	supply_consumption = 0.02	# Supply use - How much supply a unit consumes per day
									#补给使用率-一个单位每天消耗多少补给
	default_morale = 0.3		# Recovery Rate - How much extra organisation can be regained hourly when not in combat
									# 恢复率-非战斗中可以每小时额外恢复组织的多少
	combat_width = 0.5				# 战斗宽度（不用改）
</pre>

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Costs（成本） >>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

<pre>
	lend_lease_cost = 1		# Space taken up in convoy
								# 海运时间成本
	build_cost_ic = 0.4		# Production Cost - How much factory output this piece of equipment needs
								# 生产成本-这件设备需要多少资源量
	manpower = 300			# Manpower - Cost in manpower to produce
								# 人力成本-人工生产成本
	training_time = 120		# Training time - Time in days to train this unit
								# 训练时间-每日训练单位数量
</pre>

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Offensive（进攻类） >>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

<pre>
	attack = 0.1		# Firepower - Amount of damage done per attack
							# 火力-每次攻击所造成的伤害
	soft_attack = -0.1	# Soft Attack - How many attacks the unit can make versus enemies with low hardness
							# 软攻-对低硬度敌人单位能制造多少攻击力
	hard_attack = -0.5	# Hard Attack - How many attacks the unit can make versus enemies with high hardness
							# 硬攻-对高硬度敌人单位能制造多少攻击力
	air_attack = 1		# Air Attack - How much damage we can do against airplanes. High Air Attack also helps to counter enemy Air Superiority effects
							# 防空-我们能对飞机造成多大的损失？。高空攻击也有助于对抗敌人空中优势的影响
	ap_attack = 1		# Piercing - Having equal or greater Piercing to the targets Armor value allows you to do more damage.
							# 穿甲-对目标护甲值有同等或更大的穿透力，可以造成更多的伤害。
	breakthrough = 0.5	# Breakthrough - How many enemy attacks a unit can attempt to avoid while on the offensive, effectively allowing it to stay on the offense longer.
							# 突破-在进攻时避免多个敌人单位进攻，有效地使其延长进攻时间。
</pre>

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Defensive（防御类） >>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>

<pre>
	defense = 0.1		# Defense - How many enemy attacks a unit can avoid whilst on the defensive, effectively allowing it to stay on the defensive longer.
							# 防御-在防御时避免多个敌人单位进攻，有效地使其延长进攻时间。
	max_strength = 2	# HP - Strength represents how much damage this unit can suffer before it is destroyed
							# 最大强度（HP）-代表人力被消灭前所承受的伤害
	armor_value = 0		# Armor - Armor that is higher than the opponents Piercing value reduces damage taken and allows more attacks to occur
							# 装甲-高于对手穿透值减少伤害，并顶住更多攻击
	hardness = 0.5		# Hardness - Represents how much of your divsion is made up of armoured vehicles. High Hardness = High Hard Attacks, Low Soft Attack
							# 硬度-代表你有多少数值是由装甲载具提供。高硬度=高硬攻击，低软攻
	entrenchment = 5	# Entrenchment - The ability to make proper defensive entrenchments before a hostile attack

# The two types of events are news_event and country_event.
# news_event is generally used to notify players about stuff.
# country_event is generally used to make the player/ai decide something.
# The Meantime to Happen, usually referred to as MTTH,
# states the average time it will take for the event to occur
	# 这两种类型的事件news_event和country_event。
	# news_event通常用来通知全员有关的东西。
	# country_event一般是用来让玩家与AI决定某事。
	# 同时发生的，通常称为MTTH，
	# 说明事件发生的平均时间。

	mean_time_to_happen = {
		days = 500
		modifier = {
			factor = 0.4						# If true multiply 500 by 0.4 = 200
													# 如果成立则500乘以0.4等于200
			NOT = { country_exists = SOV }
		}
		modifier = {
			factor = 0.4						# If true multiply 200 by 0.4 = 40
													# 如果成立则200乘以0.4等于40
			NOT = { country_exists = ENG }
		}
		modifier = {
			factor = 0.1						# If true multiply 40 by 0.1 = 4
													# 如果成立则40乘以0.1等于4
			NOT = { country_exists = ENG }
			NOT = { country_exists = SOV }
		}
	}

# The Trigger block is what is required for the event to fire. The event console command ignores this block.

		country_exists = USA
	}

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Examples（实例） <<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<

例如，下面的命令将导致游戏中的每个国家举行选举：

<pre>
	every_country = {
		hold_election = yes
}
</pre>

# while this block will cause an election in every country with a democratic government:
	# 而这个限制将导致每个国家都会民主政府选举

<pre>
	every_country = {
		limit = {
			has_government = democratic
		}
		hold_election = yes
	}
</pre>

# While this simply shortens the list of entities affected, limits are also often used to narrow a random select:
	# 虽然仅缩减受影响的群体，但是通常也缩小了随机选择的界限

<pre>
	random_country = {
		limit = {
			NOT = { has_war = yes }
		}
		# Do effect
			# 效果
	}
</pre>

这将选择一个没有战争的随机国家。
（注意：不是<code>NOT = { has_war = yes }</code>当然也可能只是说<code>has_war = no</code>。
这实际上是“''better '，但这种方式能够展示如何做一个简单的否定，与任何其他情况，并不是所有的都是反像这样简单的）。

限制本身也可以使用作用域，但不是所有命令块中有效的作用域在条件作用域中也是有效的。
例如<code>every_country</code>可以使用的命令，但条件<code>any_country</code>必须代替。
如果至少有一个有民主政府的国家存在，该法典将导致在每个国家举行选举：

<pre>
	if = {
		limit = {
			any_country = {
				has_government = democratic
			}
		}
		every_country = {
			hold_election = yes
		}
	}
</pre>

请注意在本例中使用<code>if</code>。简单地放在一个正常的命令作用域中会限制命令应用到的实体，但是命令总是被执行。
另一方面，如<code>limit</code>代码确定了命令是否执行：如果<code>if</code>计算为<code>true</code>，则执行命令。否则，它们将被忽略。

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Operators（运行） <<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<

虽然使用单一条件可以侥幸成功，但通常真正需要的是检查一组复杂的条件，这些条件必须存在于任何特定场景中才能应用。
基本上，当问“战争的主题是什么？”很简单，通常你需要一个过滤器，比如“A与B打赢”，或者“A是C阵营，还是和B失去一场战争”？或者可能是“C与B联合，与A作战”？

为此，有一组简单但功能强大的工具，作为普通逻辑语法的一部分：

# AND
# All conditions contained within must be true
	# 所包含的所有条件必须是真实的。

		AND = {
			ai_irrationality > 50
			has_war = yes
		}

# OR
# At least one of the following conditions must be true
	# 至少下列条件之一必须是真实的

		OR = {
			is_coastal = yes
			free_building_slots > 5
		}

# NOT
# Reverses the logic result of its content. That is, if the content is "True", then the result of the NOT is "False" and vice versa.
	# 颠倒内容的逻辑结果。也就是说，如果内容为“true”，则“不”的结果是“false”，反之亦然。

		NOT = {
			has_global_flag = fall_of_france
		}

如果没有使用这些代码，则假定<code>AND</code>；显式的<code>AND</code>作用域最常用于<code>OR</code>范围内，或者解释其他难以读取的代码。

更棒的是，你可以把它们嵌套在一起，以建立一个难以理解的灵活性的真正复杂的条件。

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>> Scope-change types（范围变更类型） <<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<
#There are different types of scope change operators.
	不同类型的范围变更操作符

	* Condition: Conditional scopes do return a Boolean value like other conditions
		* 条件：与其他条件一样的布尔数值将返回条件作用范围
	* Command: Command scopes do execute commands on the new scope
		* 指令：指令作用将在新范围内执行
	* Both: These operators can be used as both of the above.
		* 二者：以上两种运算符都可使用

# Most of the general scope change operators have four different forms.
	# 在多数一般变化范围内有四种不同的执行方式

# all_<name>
	# Conditional, all must meet the subconditions
		# 所有 都必须满足附加条件的执行

# any_<name>
	# Conditional, at least one must meet the subconditions
		# 任何 至少满足一个附加条件的执行

# every_<name>
	# Command, executed on every subscope
		# 每个 对每个子范围执行

# random_<name>
	# Command, executed on one random subscope
		# 随机 在一个子范围内执行
